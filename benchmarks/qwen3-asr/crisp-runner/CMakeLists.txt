cmake_minimum_required(VERSION 3.20)
project(qwen_crisp_bench LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CRISP_SOURCE)
 message(FATAL_ERROR "Specify pinned external CRISP_SOURCE")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option GGML_CUDA GGML_HIP GGML_VULKAN GGML_METAL GGML_SYCL GGML_RPC GGML_OPENCL GGML_BUILD_TESTS GGML_BUILD_EXAMPLES GGML_BACKEND_DL GGML_NATIVE)
 set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_OPENMP ON CACHE BOOL "" FORCE)
add_subdirectory(${CRISP_SOURCE}/ggml ggml)
find_package(Threads REQUIRED)
add_library(crispasr-core STATIC ${CRISP_SOURCE}/src/core/gguf_loader.cpp ${CRISP_SOURCE}/src/core/mel.cpp ${CRISP_SOURCE}/src/crispasr_imatrix.cpp)
target_include_directories(crispasr-core PUBLIC ${CRISP_SOURCE}/src ${CRISP_SOURCE}/ggml/src)
target_link_libraries(crispasr-core PUBLIC ggml ggml-base ggml-cpu Threads::Threads ${CMAKE_DL_LIBS})
add_subdirectory(${CRISP_SOURCE}/crisp_audio crisp_audio)
add_library(qwen3_asr STATIC ${CRISP_SOURCE}/src/qwen3_asr.cpp)
target_include_directories(qwen3_asr PUBLIC ${CRISP_SOURCE}/src)
target_link_libraries(qwen3_asr PUBLIC crispasr-core crisp_audio ggml ggml-cpu)
add_executable(qwen-crisp-bench main.cpp)
target_link_libraries(qwen-crisp-bench PRIVATE qwen3_asr)
